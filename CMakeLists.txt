cmake_minimum_required(VERSION 3.21)
project(OpenTrackStudio VERSION 0.1.6 LANGUAGES CXX RC)
set(CMAKE_AUTORCC ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "This preview targets Windows (the track codec uses Windows BCrypt).")
endif()
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Concurrent OpenGL OpenGLWidgets Network)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
endif()
add_library(trackcore STATIC src/document.cpp src/recognizer.cpp src/velocidrone.cpp src/aivision.cpp src/imageimport.cpp)
target_include_directories(trackcore PUBLIC src)
target_link_libraries(trackcore PUBLIC Qt6::Gui Qt6::Network PRIVATE bcrypt advapi32)
qt_add_executable(OpenTrackStudio WIN32 src/main.cpp src/window.cpp src/canvas.cpp src/scene3d.cpp src/aidialog.cpp assets/resources.qrc assets/app.rc)
target_include_directories(OpenTrackStudio PRIVATE assets)
target_link_libraries(OpenTrackStudio PRIVATE trackcore Qt6::Widgets Qt6::Concurrent Qt6::OpenGLWidgets Qt6::OpenGL)
enable_testing()
foreach(component document color geometry route recognizer velocidrone aivision imageimport)
  add_executable(${component}_test tests/${component}_test.cpp)
  target_link_libraries(${component}_test PRIVATE trackcore)
  add_test(NAME ${component} COMMAND ${component}_test)
endforeach()
add_executable(canvas_test tests/canvas_test.cpp src/canvas.cpp)
target_link_libraries(canvas_test PRIVATE trackcore Qt6::Widgets)
add_test(NAME canvas COMMAND canvas_test)
set_tests_properties(canvas PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
add_executable(scene3d_test tests/scene3d_test.cpp src/scene3d.cpp)
target_link_libraries(scene3d_test PRIVATE trackcore Qt6::OpenGLWidgets Qt6::OpenGL)
add_test(NAME scene3d COMMAND scene3d_test)
set_tests_properties(scene3d PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
add_test(NAME app_smoke COMMAND OpenTrackStudio --smoke-test "${CMAKE_CURRENT_BINARY_DIR}/smoke")
set_tests_properties(app_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
install(TARGETS OpenTrackStudio RUNTIME DESTINATION .)
